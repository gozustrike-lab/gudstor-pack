    42→// ─── Related Products Carousel ──────────────────────────────────────────
    43→function RelatedProductsCarousel({ products: rpList }: { products: typeof visibleProducts }) {
    44→  const scrollRef = useRef<HTMLDivElement>(null);
    45→  const [canLeft, setCanLeft] = useState(false);
    46→  const [canRight, setCanRight] = useState(true);
    47→  const isDragging = useRef(false);
    48→  const dragStartX = useRef(0);
    49→  const dragScrollStart = useRef(0);
    50→
    51→  const updateBtns = useCallback(() => {
    52→    const el = scrollRef.current;
    53→    if (!el) return;
    54→    setCanLeft(el.scrollLeft > 5);
    55→    setCanRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 5);
    56→  }, []);
    57→
    58→  useEffect(() => {
    59→    const el = scrollRef.current;
    60→    if (!el) return;
    61→    updateBtns();
    62→    el.addEventListener('scroll', updateBtns, { passive: true });
    63→    const ro = new ResizeObserver(updateBtns);
    64→    ro.observe(el);
    65→    return () => { el.removeEventListener('scroll', updateBtns); ro.disconnect(); };
    66→  }, [updateBtns]);
    67→
    68→  const scroll = useCallback((dir: 'left' | 'right') => {
    69→    const el = scrollRef.current;
    70→    if (!el) return;
    71→    const cardW = el.querySelector<HTMLElement>('[data-rp-card]')?.offsetWidth || 160;
    72→    const gap = 12;
    73→    el.scrollBy({ left: dir === 'left' ? -(cardW + gap) : (cardW + gap), behavior: 'smooth' });
    74→  }, []);
    75→
    76→  // Touch drag handlers
    77→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
    78→    isDragging.current = true;
    79→    dragStartX.current = e.touches[0].pageX;
    80→    dragScrollStart.current = scrollRef.current?.scrollLeft || 0;
    81→  }, []);
    82→
    83→  const handleTouchMove = useCallback((e: React.TouchEvent) => {
    84→    if (!isDragging.current || !scrollRef.current) return;
    85→    const dx = e.touches[0].pageX - dragStartX.current;
    86→    scrollRef.current.scrollLeft = dragScrollStart.current - dx;
    87→  }, []);
    88→
    89→  const handleTouchEnd = useCallback(() => {
    90→    isDragging.current = false;
    91→  }, []);
    92→
    93→  return (
    94→    <section className="mt-8 sm:mt-12 pt-6 border-t border-border/50">
    95→      <div className="flex items-center justify-between mb-3 sm:mb-4">
    96→        <h2 className="text-base sm:text-lg font-bold text-foreground">Productos Relacionados</h2>
    97→        <div className="flex items-center gap-1.5">
    98→          <button
    99→            onClick={() => scroll('left')}
   100→            disabled={!canLeft}
   101→            className="w-8 h-8 flex items-center justify-center rounded-full border border-border/50 bg-white hover:bg-muted/50 disabled:opacity-30 disabled:pointer-events-none transition-all"
   102→            aria-label="Anterior"
   103→          >
   104→            <ChevronLeft className="w-4 h-4 text-foreground/70" />
   105→          </button>
   106→          <button
   107→            onClick={() => scroll('right')}
   108→            disabled={!canRight}
   109→            className="w-8 h-8 flex items-center justify-center rounded-full border border-border/50 bg-white hover:bg-muted/50 disabled:opacity-30 disabled:pointer-events-none transition-all"
   110→            aria-label="Siguiente"
   111→          >
   112→            <ChevronRight className="w-4 h-4 text-foreground/70" />
   113→          </button>
   114→        </div>
   115→      </div>
   116→      <div
   117→        ref={scrollRef}
   118→        className="flex gap-3 overflow-x-auto pb-2 scrollbar-hide cursor-grab active:cursor-grabbing"
   119→        onTouchStart={handleTouchStart}
   120→        onTouchMove={handleTouchMove}
   121→        onTouchEnd={handleTouchEnd}
   122→      >
   123→        {rpList.map((rp) => (
   124→          <Link
   125→            key={rp.id}
   126→            href={`/productos/${rp.slug || rp.id}`}
   127→            className="shrink-0 w-[160px] sm:w-[200px]"
   128→          >
   129→            <motion.div
   130→              initial={{ opacity: 0, y: 15 }}
   131→              whileInView={{ opacity: 1, y: 0 }}
   132→              viewport={{ once: true }}
   133→              className="group bg-card border border-border/50 rounded-xl sm:rounded-2xl overflow-hidden hover:shadow-md transition-shadow"
   134→            >
   135→              <div data-rp-card className="aspect-[4/3] bg-gradient-to-br from-muted to-muted/50 flex items-center justify-center">
   136→                <Package className="w-8 h-8 sm:w-10 sm:h-10 text-primary/30" />
   137→              </div>
   138→              <div className="p-2.5 sm:p-3">
   139→                <h3 className="text-[11px] sm:text-xs font-bold text-foreground group-hover:text-primary truncate">{rp.nombre}</h3>
   140→                <p className="text-sm sm:text-lg font-extrabold text-primary mt-0.5 sm:mt-1">
   141→                  {formatPrice(rp.packs[0].precio / rp.packs[0].cantidad)}/u
   142→                </p>
   143→              </div>
   144→            </motion.div>
   145→          </Link>
   146→        ))}
   147→      </div>
   148→    </section>
   149→  );
   150→}
