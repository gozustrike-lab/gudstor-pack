   122→function matchSubcategory(product: (typeof visibleProducts)[0], subcat: string): boolean {
   123→  const patterns = SUBCATEGORY_TAG_MAP[subcat];
   124→  if (!patterns) return false;
   125→  const productTags = (product.etiquetas || []).map((t) => t.toLowerCase());
   126→  const nameLower = product.nombre.toLowerCase();
   127→  const materialLower = product.material.toLowerCase();
   128→
   129→  return patterns.some((pattern) => {
   130→    const p = pattern.toLowerCase();
   131→    return productTags.includes(p) || nameLower.includes(p) || materialLower.includes(p);
   132→  });
   133→}

   350→// ── Main Content ───────────────────────────────────────────────────
   351→function ProductosContent() {
   352→  const searchParams = useSearchParams();
   353→
   354→  // ── Read URL params on mount (deep linking from WhatsApp/shared links) ──
   355→  const categoriaParam = searchParams.get('categoria') || '';
   356→  const subcategoriaParam = searchParams.get('subcategoria') || '';
   357→
   358→  const router = useRouter();
   359→  const [selectedCategory, setSelectedCategory] = useState(categoriaParam);
   360→  const [selectedSubcategory, setSelectedSubcategory] = useState(subcategoriaParam);
   361→  const [sortBy, setSortBy] = useState('nombre');
   362→  const [isFilterOpen, setIsFilterOpen] = useState(false);
   363→  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid');
   364→  const isFirstRender = useRef(true);
   365→
   366→  useScrollSpy({
   367→    ids: ['catalogo', 'productos-listado', 'sidebar-filtros'],
   368→    offset: '100px',
   369→  });
   370→
   371→  // ── Sync state → URL via Next.js router (keeps internal state in sync) ──
   372→  useEffect(() => {
   373→    if (isFirstRender.current) {
   374→      isFirstRender.current = false;
   375→      return;
   376→    }
   377→    const params = new URLSearchParams();
   378→    if (selectedCategory) params.set('categoria', selectedCategory);
   379→    if (selectedSubcategory) params.set('subcategoria', selectedSubcategory);
   380→    const qs = params.toString();
   381→    router.replace(`/productos${qs ? `?${qs}` : ''}`, { scroll: false });
   382→  }, [selectedCategory, selectedSubcategory, router]);
   383→
   384→  // ── Dynamic browser title + meta description ──
   385→  useEffect(() => {
   386→    const meta = CATEGORY_META[selectedCategory] || DEFAULT_META;
   387→    const pageTitle = selectedSubcategory
   388→      ? `${selectedSubcategory} | ${selectedCategory} | GUDSTOR PACK`
   389→      : meta.title;
   390→    document.title = pageTitle;
   391→    // Update meta description
   392→    const descEl = document.querySelector('meta[name="description"]');
   393→    if (descEl) {
   394→      descEl.setAttribute('content', meta.description);
   395→    }
   396→    // Update canonical URL
   397→    const canonEl = document.querySelector('link[rel="canonical"]');
   398→    if (canonEl) {
   399→      const params = new URLSearchParams();
   400→      if (selectedCategory) params.set('categoria', selectedCategory);
   401→      if (selectedSubcategory) params.set('subcategoria', selectedSubcategory);
   402→      const qs = params.toString();
   403→      canonEl.setAttribute('href', `https://gudstor-pack.vercel.app/productos${qs ? `?${qs}` : ''}`);
   404→    }
   405→  }, [selectedCategory, selectedSubcategory]);
   406→
   407→  const filteredProducts = useMemo(() => {
   408→    let result = [...visibleProducts];
   409→
   410→    if (selectedCategory) {
   411→      result = result.filter((p) => p.categoria === selectedCategory);
   412→    }
   413→    if (selectedSubcategory && selectedCategory) {
   414→      result = result.filter((p) => matchSubcategory(p, selectedSubcategory));
   415→    }
   416→
   417→    result.sort((a, b) => {
   418→      switch (sortBy) {
   419→        case 'destacados':
   420→          return (b.destacado ? 1 : 0) - (a.destacado ? 1 : 0) || a.nombre.localeCompare(b.nombre);
   421→        case 'precio-asc':
   422→          return a.precio - b.precio;
   423→        case 'precio-desc':
   424→          return b.precio - a.precio;
   425→        case 'nombre':
   426→        default:
   427→          return a.nombre.localeCompare(b.nombre);
   428→      }
   429→    });
   430→
   431→    return result;
   432→  }, [selectedCategory, selectedSubcategory, sortBy]);
   433→
   434→  // Ref to always read latest state inside stable callbacks (prevents stale closures)
   435→  const filterRef = useRef({ selectedCategory: '', selectedSubcategory: '' });
   436→  filterRef.current = { selectedCategory, selectedSubcategory };
   437→
   438→  const handleCategoryClick = useCallback((cat: string) => {
   439→    const { selectedCategory: prev } = filterRef.current;
   440→    if (cat === prev) {
   441→      setSelectedCategory('');
   442→      setSelectedSubcategory('');
   443→    } else {
   444→      setSelectedCategory(cat);
   445→      setSelectedSubcategory('');
   446→    }
   447→  }, []);
   448→
   449→  const handleSubcategoryClick = useCallback((cat: string, sub: string) => {
   450→    const { selectedCategory: prevCat, selectedSubcategory: prevSub } = filterRef.current;
   451→    if (sub === prevSub && cat === prevCat) {
   452→      setSelectedSubcategory('');
   453→    } else {
   454→      setSelectedCategory(cat);
   455→      setSelectedSubcategory(sub);
   456→    }
   457→  }, []);
   458→
   459→  const clearAll = useCallback(() => {
   460→    setSelectedCategory('');
   461→    setSelectedSubcategory('');
   462→  }, []);
   463→
   464→  const activeFiltersCount = (selectedCategory ? 1 : 0) + (selectedSubcategory ? 1 : 0);
   465→
   466→  const breadcrumbs = useMemo(() => {
   467→    const parts: { label: string; action?: () => void }[] = [];
   468→    parts.push({ label: 'Inicio' });
   469→    parts.push({ label: 'Productos' });
   470→    if (selectedCategory) {
   471→      parts.push({
   472→        label: selectedCategory,
   473→        action: () => {
   474→          setSelectedSubcategory('');
   475→        },
   476→      });
   477→    }
   478→    if (selectedSubcategory) {
   479→      parts.push({ label: selectedSubcategory });
   480→    }
   481→    return parts;
   482→  }, [selectedCategory, selectedSubcategory]);
   483→
   484→  return (
   485→    <div className="min-h-screen bg-background">
   486→      {/* ── Immersive Hero Banner ─────────────────────────────── */}
   487→      <ImmersiveBanner
   488→        title={selectedSubcategory || (selectedCategory ? selectedCategory : 'Catálogo de Productos')}
   489→        subtitle={`${filteredProducts.length} ${filteredProducts.length === 1 ? 'producto' : 'productos'} disponibles`}
   490→        breadcrumb={breadcrumbs.map(b => ({ label: b.label, href: b.action ? '#' : undefined }))}
   491→      />
   492→
   493→      {/* ── Main Layout: Sidebar + Products ─────────────────────── */}
   494→      <div className="max-w-[1400px] mx-auto px-4 sm:px-6 lg:px-8 pt-4 sm:pt-6 pb-6 sm:pb-8">
   495→        {/* Toolbar: Filter + Sort + View Mode — sticky below banner */}
   496→        <div id="catalogo" className="flex items-center gap-2 mb-4 sm:mb-6 -mx-4 sm:mx-0 px-4 sm:px-0">
   497→          {/* Filter toggle (mobile) */}
   498→          <button
   499→            onClick={() => setIsFilterOpen(!isFilterOpen)}
   500→            className={`lg:hidden inline-flex items-center gap-2 px-4 py-2.5 rounded-xl text-sm font-semibold transition-all ${
   501→              isFilterOpen
   502→                ? 'bg-primary text-primary-foreground shadow-sm shadow-primary/20'
   503→                : 'bg-card border border-border hover:bg-muted/50'
   504→            }`}
   505→          >
   506→            <SlidersHorizontal className="w-4 h-4" />
   507→            <span>Categorías</span>
   508→            {activeFiltersCount > 0 && (
   509→              <span className="w-5 h-5 bg-primary text-primary-foreground text-[10px] font-bold rounded-full flex items-center justify-center">
   510→                {activeFiltersCount}
   511→              </span>
   512→            )}
   513→          </button>
   514→
   515→          <div className="flex-1" />
   516→
   517→          {/* Sort */}
   518→          <div className="relative">
   519→            <select
   520→              value={sortBy}
   521→              onChange={(e) => setSortBy(e.target.value)}
   522→              className="appearance-none bg-card border border-border rounded-xl px-4 py-2.5 pr-8 text-sm font-medium outline-none focus:border-primary cursor-pointer"
   523→            >
   524→              <option value="destacados">Más vendidos</option>
   525→              <option value="nombre">Nombre A-Z</option>
   526→              <option value="precio-asc">Precio: Menor a Mayor</option>
   527→              <option value="precio-desc">Precio: Mayor a Menor</option>
   528→            </select>
   529→            <ChevronDown className="absolute right-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground pointer-events-none" />
   530→          </div>
   531→
   532→          {/* View Mode */}
   533→          <div className="hidden sm:flex items-center bg-card border border-border rounded-xl overflow-hidden">
   534→            <button
   535→              onClick={() => setViewMode('grid')}
   536→              className={`p-2.5 transition-colors ${viewMode === 'grid' ? 'bg-primary/10 text-primary' : 'text-muted-foreground hover:text-foreground'}`}
   537→            >
   538→              <Grid3X3 className="w-4 h-4" />
   539→            </button>
   540→            <button
   541→              onClick={() => setViewMode('list')}
   542→              className={`p-2.5 transition-colors ${viewMode === 'list' ? 'bg-primary/10 text-primary' : 'text-muted-foreground hover:text-foreground'}`}
   543→            >
   544→              <List className="w-4 h-4" />
   545→            </button>
   546→          </div>
   547→        </div>
   548→
   549→        <div className="flex gap-6 lg:gap-8">
   550→          {/* ── Desktop Sidebar (sticky, always visible, professional) ── */}
   551→          <aside id="sidebar-filtros" className="hidden lg:block w-[280px] xl:w-[300px] shrink-0">
   552→            <div
   553→              className="sticky top-[96px] bg-card border border-border/60 rounded-2xl shadow-sm shadow-black/[0.03] overflow-hidden"
   554→            >
   555→              <div className="p-5 max-h-[calc(100vh-112px)] overflow-y-auto" style={{ scrollbarWidth: 'thin' }}>
   556→                <CategorySidebar
   557→                  selectedCategory={selectedCategory}
   558→                  selectedSubcategory={selectedSubcategory}
   559→                  onCategoryClick={handleCategoryClick}
   560→                  onSubcategoryClick={handleSubcategoryClick}
   561→                  onClearAll={clearAll}
   562→                />
   563→              </div>
   564→            </div>
   565→          </aside>
   566→
   567→          {/* ── Mobile Filter Drawer (full-height, professional) ──── */}
   568→          <AnimatePresence>
   569→            {isFilterOpen && (
   570→              <>
   571→                <motion.div
   572→                  initial={{ opacity: 0 }}
   573→                  animate={{ opacity: 1 }}
   574→                  exit={{ opacity: 0 }}
   575→                  transition={{ duration: 0.2 }}
   576→                  className="fixed inset-0 z-40 bg-black/50 backdrop-blur-sm lg:hidden"
   577→                  onClick={() => setIsFilterOpen(false)}
   578→                />
   579→                <motion.div
   580→                  initial={{ x: '-100%' }}
   581→                  animate={{ x: 0 }}
   582→                  exit={{ x: '-100%' }}
   583→                  transition={{ type: 'spring', stiffness: 320, damping: 32 }}
   584→                  className="fixed top-0 left-0 bottom-0 z-50 w-[300px] max-w-[85vw] bg-white shadow-2xl lg:hidden flex flex-col"
   585→                >
   586→                  <CategorySidebar
   587→                    selectedCategory={selectedCategory}
   588→                    selectedSubcategory={selectedSubcategory}
   589→                    onCategoryClick={handleCategoryClick}
   590→                    onSubcategoryClick={handleSubcategoryClick}
   591→                    onClearAll={clearAll}
   592→                    isMobile
   593→                    onClose={() => setIsFilterOpen(false)}
   594→                  />
   595→                </motion.div>
   596→              </>
   597→            )}
   598→          </AnimatePresence>
   599→
   600→          {/* ── Product Grid ────────────────────────────────────── */}
   601→          <div id="productos-listado" className="flex-1 min-w-0">
   602→            {/* Active filter tags */}
   603→            {activeFiltersCount > 0 && (
   604→              <div className="flex flex-wrap gap-2 mb-4">
   605→                {selectedCategory && (
   606→                  <span className="inline-flex items-center gap-1 px-3 py-1 bg-primary/10 text-primary text-xs font-medium rounded-full">
   607→                    {selectedCategory}
   608→                    <button onClick={() => { setSelectedCategory(''); setSelectedSubcategory(''); }}>
   609→                      <X className="w-3 h-3" />
   610→                    </button>
   611→                  </span>
   612→                )}
   613→                {selectedSubcategory && (
   614→                  <span className="inline-flex items-center gap-1 px-3 py-1 bg-secondary/10 text-secondary text-xs font-medium rounded-full">
   615→                    {selectedSubcategory}
   616→                    <button onClick={() => setSelectedSubcategory('')}>
   617→                      <X className="w-3 h-3" />
   618→                    </button>
   619→                  </span>
   620→                )}
   621→              </div>
   622→            )}
   623→
   624→            {filteredProducts.length > 0 ? (
   625→              <div
   626→                className={
   627→                  viewMode === 'grid'
   628→                    ? 'grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 2xl:grid-cols-3 gap-4 sm:gap-5'
   629→                    : 'space-y-4'
   630→                }
   631→              >
   632→                <AnimatePresence mode="wait">
   633→                  {filteredProducts.map((product, i) => (
   634→                    <ProductCard key={product.id} product={product} index={i} />
   635→                  ))}
   636→                </AnimatePresence>
   637→              </div>
   638→            ) : (
   639→              <motion.div
   640→                initial={{ opacity: 0 }}
   641→                animate={{ opacity: 1 }}
   642→                className="flex flex-col items-center justify-center py-20 text-center"
   643→              >
   644→                <div className="w-20 h-20 bg-muted rounded-2xl flex items-center justify-center mb-4">
   645→                  <Package className="w-10 h-10 text-muted-foreground" />
   646→                </div>
   647→                <h3 className="text-lg font-bold text-foreground mb-2">Sin resultados</h3>
   648→                <p className="text-sm text-muted-foreground max-w-sm">
   649→                  No se encontraron productos con los filtros seleccionados. Prueba con otra categoría o
   650→                  subcategoría.
   651→                </p>
   652→                <button
   653→                  onClick={clearAll}
   654→                  className="mt-4 px-6 py-2.5 bg-primary text-primary-foreground text-sm font-semibold rounded-xl hover:bg-primary/90 transition-colors"
   655→                >
   656→                  Ver todos los productos
   657→                </button>
   658→              </motion.div>
   659→            )}
   660→          </div>
   661→        </div>
   662→      </div>
   663→    </div>
   664→  );
   665→}
