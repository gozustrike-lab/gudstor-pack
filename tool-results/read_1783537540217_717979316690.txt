   279→
   280→function ProductSlider() {
   281→  const fp = visibleProducts.filter((p) => p.destacado);
   282→  const dup = [...fp, ...fp];
   283→  const scrollRef = useRef<HTMLDivElement>(null);
   284→  const [paused, setPaused] = useState(false);
   285→
   286→  useEffect(() => {
   287→    const el = scrollRef.current;
   288→    if (!el || paused) return;
   289→    let pos = 0;
   290→    let raf: number;
   291→    const step = () => { pos += 0.8; if (pos >= el.scrollWidth / 2) pos = 0; el.scrollLeft = pos; raf = requestAnimationFrame(step); };
   292→    raf = requestAnimationFrame(step);
   293→    return () => cancelAnimationFrame(raf);
   294→  }, [paused]);
   295→
   296→  return (
   297→    <div className="relative group" onMouseEnter={() => setPaused(true)} onMouseLeave={() => setPaused(false)}>
   298→      <div className="absolute left-0 top-0 bottom-0 w-12 bg-gradient-to-r from-muted/30 to-transparent z-10 pointer-events-none" />
   299→      <div className="absolute right-0 top-0 bottom-0 w-12 bg-gradient-to-l from-muted/30 to-transparent z-10 pointer-events-none" />
   300→      <div ref={scrollRef} className="flex gap-4 sm:gap-6 overflow-hidden" style={{ scrollbarWidth: 'none' }}>
   301→        {dup.map((p, i) => (<div key={`${p.id}-${i}`} className="shrink-0" style={{ width: 320 }}><ProductCard product={p} index={i % fp.length} /></div>))}
   302→      </div>
   303→    </div>
   304→  );
   305→}
