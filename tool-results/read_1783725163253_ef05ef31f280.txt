   279→function ProductSlider() {
   280→  const fp = visibleProducts.filter((p) => p.destacado);
   281→  const dup = [...fp, ...fp];
   282→  const scrollRef = useRef<HTMLDivElement>(null);
   283→  const [paused, setPaused] = useState(false);
   284→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   285→  const [canScrollRight, setCanScrollRight] = useState(true);
   286→  const isDragging = useRef(false);
   287→  const startX = useRef(0);
   288→  const scrollStart = useRef(0);
   289→  const pausedRef = useRef(false);
   290→
   291→  // Keep ref in sync
   292→  useEffect(() => { pausedRef.current = paused; }, [paused]);
   293→
   294→  // Auto-scroll animation
   295→  useEffect(() => {
   296→    const el = scrollRef.current;
   297→    if (!el || paused) return;
   298→    let pos = el.scrollLeft;
   299→    let raf: number;
   300→    const step = () => {
   301→      if (pausedRef.current) return;
   302→      pos += 0.8;
   303→      if (pos >= el.scrollWidth / 2) pos = 0;
   304→      el.scrollLeft = pos;
   305→      raf = requestAnimationFrame(step);
   306→    };
   307→    raf = requestAnimationFrame(step);
   308→    return () => cancelAnimationFrame(raf);
   309→  }, [paused]);
   310→
   311→  // Pause on interaction, resume after 4s
   312→  const pauseImmediately = useCallback(() => {
   313→    pausedRef.current = true;
   314→    setPaused(true);
   315→  }, []);
   316→
   317→  useEffect(() => {
   318→    if (!paused) return;
   319→    const timer = setTimeout(() => { pausedRef.current = false; setPaused(false); }, 4000);
   320→    return () => clearTimeout(timer);
   321→  }, [paused]);
   322→
   323→  // Update scroll button visibility
   324→  const updateScrollButtons = useCallback(() => {
   325→    const el = scrollRef.current;
   326→    if (!el) return;
   327→    setCanScrollLeft(el.scrollLeft > 5);
   328→    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 5);
   329→  }, []);
   330→
   331→  useEffect(() => {
   332→    const el = scrollRef.current;
   333→    if (!el) return;
   334→    updateScrollButtons();
   335→    el.addEventListener('scroll', updateScrollButtons, { passive: true });
   336→    return () => el.removeEventListener('scroll', updateScrollButtons);
   337→  }, [updateScrollButtons]);
   338→
   339→  // Drag to scroll (mouse)
   340→  const handleMouseDown = useCallback((e: React.MouseEvent) => {
   341→    isDragging.current = true;
   342→    startX.current = e.pageX - scrollRef.current!.offsetLeft;
   343→    scrollStart.current = scrollRef.current!.scrollLeft;
   344→    pauseImmediately();
   345→  }, [pauseImmediately]);
   346→
   347→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   348→    if (!isDragging.current) return;
   349→    e.preventDefault();
   350→    const x = e.pageX - scrollRef.current!.offsetLeft;
   351→    const walk = (x - startX.current) * 1.5;
   352→    scrollRef.current!.scrollLeft = scrollStart.current - walk;
   353→  }, []);
   354→
   355→  const handleMouseUp = useCallback(() => {
   356→    isDragging.current = false;
   357→  }, []);
   358→
   359→  const handleMouseLeave = useCallback(() => {
   360→    isDragging.current = false;
   361→  }, []);
   362→
   363→  // Drag to scroll (touch)
   364→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   365→    isDragging.current = true;
   366→    startX.current = e.touches[0].pageX - scrollRef.current!.offsetLeft;
   367→    scrollStart.current = scrollRef.current!.scrollLeft;
   368→    pauseImmediately();
   369→  }, [pauseImmediately]);
   370→
   371→  const handleTouchMove = useCallback((e: React.TouchEvent) => {
   372→    if (!isDragging.current) return;
   373→    const x = e.touches[0].pageX - scrollRef.current!.offsetLeft;
   374→    const walk = (x - startX.current) * 1.2;
   375→    const el = scrollRef.current!;
   376→    el.scrollLeft = scrollStart.current - walk;
   377→  }, []);
   378→
   379→  const handleTouchEnd = useCallback(() => {
   380→    isDragging.current = false;
   381→  }, []);
   382→
   383→  // Arrow navigation
   384→  const navigateSlider = useCallback((direction: 'left' | 'right') => {
   385→    const el = scrollRef.current;
   386→    if (!el) return;
   387→    const cardWidth = 320 + 24; // card width + gap
   388→    el.scrollBy({ left: direction === 'left' ? -cardWidth : cardWidth, behavior: 'smooth' });
   389→    setPaused(true);
   390→  }, []);
   391→
   392→  return (
   393→    <div
   394→      className="relative group"
   395→      onMouseEnter={() => setPaused(true)}
   396→      onMouseLeave={() => { setPaused(false); handleMouseUp(); }}
   397→      onTouchStart={handleTouchStart}
   398→      onTouchMove={handleTouchMove}
   399→      onTouchEnd={handleTouchEnd}
   400→      onClick={pauseImmediately}
   401→      onMouseDown={handleMouseDown}
   402→      onMouseMove={handleMouseMove}
   403→      onMouseUp={handleMouseUp}
   404→      onMouseLeaveCapture={handleMouseLeave}
   405→    >
   406→      {/* Left Arrow */}
   407→      {canScrollLeft && (
   408→        <button
   409→          onClick={(e) => { e.stopPropagation(); navigateSlider('left'); }}
   410→          className="absolute left-1 top-1/2 -translate-y-1/2 z-20 w-9 h-9 sm:w-10 sm:h-10 flex items-center justify-center bg-white/90 backdrop-blur-sm rounded-full shadow-lg border border-border/50 hover:bg-white hover:shadow-xl transition-all md:opacity-0 md:group-hover:opacity-100"
