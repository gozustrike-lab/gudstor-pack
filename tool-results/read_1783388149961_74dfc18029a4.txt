   212→
   213→function ProductSlider() {
   214→  const fp = visibleProducts.filter((p) => p.destacado);
   215→  const dup = [...fp, ...fp];
   216→  const scrollRef = useRef<HTMLDivElement>(null);
   217→  const [paused, setPaused] = useState(false);
   218→
   219→  useEffect(() => {
   220→    const el = scrollRef.current;
   221→    if (!el || paused) return;
   222→    let pos = 0;
   223→    let raf: number;
   224→    const step = () => { pos += 0.8; if (pos >= el.scrollWidth / 2) pos = 0; el.scrollLeft = pos; raf = requestAnimationFrame(step); };
   225→    raf = requestAnimationFrame(step);
   226→    return () => cancelAnimationFrame(raf);
   227→  }, [paused]);
   228→
   229→  return (
   230→    <div className="relative group" onMouseEnter={() => setPaused(true)} onMouseLeave={() => setPaused(false)}>
   231→      <div className="absolute left-0 top-0 bottom-0 w-12 bg-gradient-to-r from-muted/30 to-transparent z-10 pointer-events-none" />
   232→      <div className="absolute right-0 top-0 bottom-0 w-12 bg-gradient-to-l from-muted/30 to-transparent z-10 pointer-events-none" />
   233→      <div ref={scrollRef} className="flex gap-4 sm:gap-6 overflow-hidden" style={{ scrollbarWidth: 'none' }}>
   234→        {dup.map((p, i) => (<div key={`${p.id}-${i}`} className="shrink-0" style={{ width: 320 }}><ProductCard product={p} index={i % fp.length} /></div>))}
   235→      </div>
   236→    </div>
   237→  );
   238→}
