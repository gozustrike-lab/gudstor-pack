    42→// ─── Related Products Carousel (Auto-Scroll) ────────────────────────────
    43→function RelatedProductsCarousel({ products: rpList }: { products: typeof visibleProducts }) {
    44→  const trackRef = useRef<HTMLDivElement>(null);
    45→  const [isPaused, setIsPaused] = useState(false);
    46→  const [direction, setDirection] = useState<'left' | 'right'>('right');
    47→  const animFrameRef = useRef<number>(0);
    48→  const posRef = useRef(0);
    49→  const speedRef = useRef(0.5); // px per frame (~30px/sec at 60fps)
    50→
    51→  // Duplicate list for seamless infinite loop
    52→  const items = [...rpList, ...rpList];
    53→
    54→  // Auto-scroll animation loop
    55→  useEffect(() => {
    56→    if (isPaused) {
    57→      cancelAnimationFrame(animFrameRef.current);
    58→      return;
    59→    }
    60→    const step = () => {
    61→      const el = trackRef.current;
    62→      if (!el) return;
    63→      const halfWidth = el.scrollWidth / 2;
    64→      if (direction === 'right') {
    65→        posRef.current += speedRef.current;
    66→        if (posRef.current >= halfWidth) posRef.current -= halfWidth;
    67→      } else {
    68→        posRef.current -= speedRef.current;
    69→        if (posRef.current <= 0) posRef.current += halfWidth;
    70→      }
    71→      el.style.transform = `translateX(-${posRef.current}px)`;
    72→      animFrameRef.current = requestAnimationFrame(step);
    73→    };
    74→    animFrameRef.current = requestAnimationFrame(step);
    75→    return () => cancelAnimationFrame(animFrameRef.current);
    76→  }, [isPaused, direction]);
    77→
    78→  // Desktop: pause on hover
    79→  const handleMouseEnter = useCallback(() => setIsPaused(true), []);
    80→  const handleMouseLeave = useCallback(() => setIsPaused(false), []);
    81→
    82→  // Mobile: pause on touch, resume after delay
    83→  const handleTouchStart = useCallback(() => setIsPaused(true), []);
    84→  const handleTouchEnd = useCallback(() => {
    85→    setTimeout(() => setIsPaused(false), 2000);
    86→  }, []);
    87→
    88→  // Direction controls
    89→  const scrollDir = useCallback((dir: 'left' | 'right') => {
    90→    setDirection(dir);
    91→    setIsPaused(false);
    92→  }, []);
    93→
    94→  const toggleDirection = useCallback(() => {
    95→    setDirection(prev => prev === 'right' ? 'left' : 'right');
    96→    setIsPaused(false);
    97→  }, []);
    98→
    99→  if (rpList.length === 0) return null;
   100→
   101→  return (
   102→    <section className="mt-8 sm:mt-12 pt-6 border-t border-border/50">
   103→      <div className="flex items-center justify-between mb-3 sm:mb-4">
   104→        <h2 className="text-base sm:text-lg font-bold text-foreground">Productos Relacionados</h2>
   105→        <div className="flex items-center gap-1.5">
   106→          {/* Left button */}
   107→          <button
   108→            onClick={() => scrollDir('left')}
   109→            className={`w-8 h-8 flex items-center justify-center rounded-full border transition-all ${
   110→              direction === 'left'
   111→                ? 'border-primary bg-primary/10 text-primary'
   112→                : 'border-border/50 bg-white hover:bg-muted/50 text-foreground/70'
   113→            }`}
   114→            aria-label="Mover a la izquierda"
   115→          >
   116→            <ChevronLeft className="w-4 h-4" />
   117→          </button>
   118→          {/* Toggle auto-scroll direction */}
   119→          <button
   120→            onClick={toggleDirection}
   121→            className="px-2.5 h-8 flex items-center justify-center rounded-full border border-border/50 bg-white hover:bg-muted/50 transition-all"
   122→            aria-label="Cambiar dirección"
   123→            title={isPaused ? 'Reanudar' : `Auto-scroll: ${direction === 'right' ? '→' : '←'}`}
   124→          >
   125→            <span className={`transition-transform duration-300 inline-block ${direction === 'left' ? 'rotate-180' : ''}`}>
   126→              <ChevronRight className="w-4 h-4 text-muted-foreground" />
   127→            </span>
   128→          </button>
   129→          {/* Right button */}
   130→          <button
   131→            onClick={() => scrollDir('right')}
   132→            className={`w-8 h-8 flex items-center justify-center rounded-full border transition-all ${
   133→              direction === 'right'
   134→                ? 'border-primary bg-primary/10 text-primary'
   135→                : 'border-border/50 bg-white hover:bg-muted/50 text-foreground/70'
   136→            }`}
   137→            aria-label="Mover a la derecha"
   138→          >
   139→            <ChevronRight className="w-4 h-4" />
   140→          </button>
   141→          {/* Pause indicator dot */}
   142→          <span className={`w-1.5 h-1.5 rounded-full transition-colors ml-0.5 ${isPaused ? 'bg-amber-400' : 'bg-green-400'}`} title={isPaused ? 'Pausado' : 'En movimiento'} />
   143→        </div>
   144→      </div>
   145→
   146→      {/* Carousel viewport */}
   147→      <div
   148→        className="overflow-hidden"
   149→        onMouseEnter={handleMouseEnter}
   150→        onMouseLeave={handleMouseLeave}
   151→        onTouchStart={handleTouchStart}
   152→        onTouchEnd={handleTouchEnd}
   153→      >
   154→        <div
   155→          ref={trackRef}
   156→          className="flex gap-3 w-max will-change-transform"
   157→        >
   158→          {items.map((rp, idx) => (
   159→            <Link
   160→              key={`${rp.id}-${idx}`}
   161→              href={`/productos/${rp.slug || rp.id}`}
   162→              className="shrink-0 w-[160px] sm:w-[200px]"
   163→              onClick={() => setIsPaused(true)}
   164→            >
   165→              <motion.div
   166→                initial={{ opacity: 0, y: 15 }}
   167→                whileInView={{ opacity: 1, y: 0 }}
   168→                viewport={{ once: true }}
   169→                className="group bg-card border border-border/50 rounded-xl sm:rounded-2xl overflow-hidden hover:shadow-md transition-shadow"
   170→              >
   171→                <div data-rp-card className="aspect-[4/3] bg-gradient-to-br from-muted to-muted/50 flex items-center justify-center">
   172→                  <Package className="w-8 h-8 sm:w-10 sm:h-10 text-primary/30" />
   173→                </div>
   174→                <div className="p-2.5 sm:p-3">
   175→                  <h3 className="text-[11px] sm:text-xs font-bold text-foreground group-hover:text-primary truncate">{rp.nombre}</h3>
   176→                  <p className="text-sm sm:text-lg font-extrabold text-primary mt-0.5 sm:mt-1">
   177→                    {formatPrice(rp.packs[0].precio / rp.packs[0].cantidad)}/u
   178→                  </p>
   179→                </div>
   180→              </motion.div>
   181→            </Link>
   182→          ))}
   183→        </div>
   184→      </div>
   185→    </section>
   186→  );
   187→}
